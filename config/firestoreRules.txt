service cloud.firestore {
  match /databases/{database}/documents {
    match /{document=**} {
      allow read, write: if false;
    }
    match /pro-tuple-temp/{document=**} {
      allow read: if true;
    }
    match /pro-imagiationstat/{document=**} {
      allow read: if true;
    }
    match /pro-animals/{document=**} {
      allow read: if true;
    }
    match /pro-note/{document=**} {
      allow read: if
        request.auth != null &&
        //resource.data.author == request.auth.token.uid;
        resource.data.author == request.auth.uid;
    }
    match /pro-photocollection/{photoId} {
      allow read: if true;

      match /pro-photogps/{gpsId} {
        allow read: if true;
      }
    }
    match /pro-book_collection/{document=**} {
      allow read: if
        request.auth != null
    }
    match /pro-book/{document=**} {
      allow read: if
        request.auth != null
    }
  }
}

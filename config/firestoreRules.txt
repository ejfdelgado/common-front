rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /{document=**} {
      allow read, write: if false;
    }
    match /pro-tuple-temp/{document=**} {
      allow read: if true;
    }
    match /pro-imagiationstat/{document=**} {
      allow read: if true;
    }
    match /pro-animals/{document=**} {
      allow read: if true;
    }
    match /pro-note/{document=**} {
      allow read: if
        request.auth != null &&
        resource.data.author == request.auth.uid;
    }
    match /pro-photocollection/{photoId} {
      allow read: if true;

      match /pro-photogps/{gpsId} {
        allow read: if true;
      }
    }
    match /pro-book_collection/{document=**} {
    	allow read: if true;
    }
    match /pro-book/{document=**} {
      allow read: if true;
    }
    match /pro-document/{document=**} {
        allow read: if
        request.auth != null &&
        resource.data.author == request.auth.uid;
    }
    match /pro-knowledge/{knowledgeId} {
      allow read: if request.auth != null
                  && request.auth.uid in resource.data.owners;
    }
    match /pro-knowledge/{knowledgeId}/pro-fact/{factId} {
      allow read: if request.auth != null
                  && request.auth.uid in
                    get(/databases/$(database)/documents/pro-knowledge/$(knowledgeId))
                    .data.owners;
    }
    match /pro-knowledge/{knowledgeId}/pro-tool/{toolId} {
      allow read: if request.auth != null
                  && request.auth.uid in
                    get(/databases/$(database)/documents/pro-knowledge/$(knowledgeId))
                    .data.owners;
    }
    match /pro-knowledge/{knowledgeId}/pro-article/{articleId} {
      allow read: if request.auth != null
                  && request.auth.uid in
                    get(/databases/$(database)/documents/pro-knowledge/$(knowledgeId))
                    .data.owners;
    }
    match /pro-knowledge/{knowledgeId}/pro-history/{historyId} {
      allow read: if request.auth != null
                  && request.auth.uid in
                    get(/databases/$(database)/documents/pro-knowledge/$(knowledgeId))
                    .data.owners;
    }
    match /pro-pubknowledge/{knowledgeId} {
      allow read: if true;
    }
    match /pro-personal/{document=**} {
      allow read: if
        request.auth != null &&
        resource.data.author == request.auth.uid;
    }
    match /pro-publicpage/{document=**} {
      allow read: if true;
    }
    match /pro-client/{clientId} {
      allow read: if request.auth != null
                  && request.auth.uid in resource.data.owners;
    }
    match /pro-room-private/{document=**} {
        allow read: if
        request.auth != null && request.auth.uid in resource.data.owners;
    }
    match /pro-pug/{document=**} {
        allow read: if true;
    }
  }
}
